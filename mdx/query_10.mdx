/* ==============================================================================
   Truy vấn MDX 10: Bàn thắng chính thức của Real Madrid (Club_ID = 418) trước từng đối thủ theo quý
   Database: SSAS_Football_Transfermarkt | Cube: [DW Football Transfermarkt]
   Grain: `FACT Club Match Official` (Đội – Trận: 14.560 dòng)
   ============================================================================== */

SELECT
    {
        [Measures].[Official Team Match Count],
        [Measures].[Official Team Goals]
    } ON COLUMNS,
    ORDER(
        NONEMPTY(
            CrossJoin(
                [DIM Time].[Year Quarter Key].[Year Quarter Key].MEMBERS,
                EXCEPT(
                    [DIM Opponent Club].[Club ID].[Club ID].MEMBERS,
                    { [DIM Opponent Club].[Club ID].[Unknown] }
                )
            ),
            (
                [Measures].[Official Team Match Count],
                [DIM Club].[Club ID].&[418]
            )
        ),
        [Measures].[Official Team Goals],
        BDESC
    ) ON ROWS
FROM [DW Football Transfermarkt]
WHERE ( [DIM Club].[Club ID].&[418] );
