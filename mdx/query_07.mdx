/* ==============================================================================
   Truy vấn MDX 07: Bàn thắng theo quốc tịch cầu thủ tại các giải đấu nước Anh (và phân tích Pareto 80/20)
   Database: SSAS_Football_Transfermarkt | Cube: [DW Football Transfermarkt]
   Grain: `FACT Player Match Perf` (Lượt ra sân)
   ============================================================================== */

SELECT
    {
        [Measures].[Goals],
        [Measures].[Pct Of Country Goals]
    } ON COLUMNS,
    ORDER(
        NONEMPTY(
            EXCEPT(
                [DIM Player].[Country Of Citizenship].[Country Of Citizenship].MEMBERS,
                { [DIM Player].[Country Of Citizenship].[Unknown] }
            ),
            (
                [Measures].[Goals],
                [DIM Competition].[Country Name].&[England]
            )
        ),
        [Measures].[Goals],
        BDESC
    ) ON ROWS
FROM [DW Football Transfermarkt]
WHERE ( [DIM Competition].[Country Name].&[England] );
