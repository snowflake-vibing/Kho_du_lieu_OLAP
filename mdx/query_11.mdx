/* ==============================================================================
   Truy vấn MDX 11: Đếm số trận sân nhà của các CLB có sức chứa sân nhà trên 60.000 chỗ
   Database: SSAS_Football_Transfermarkt | Cube: [DW Football Transfermarkt]
   Grain: `DIM Game` Measure Group (Trận đấu: 7.280 dòng)
   ============================================================================== */

SELECT
    {
        [Measures].[Stadium Seats Value],
        [Measures].[Official Match Count]
    } ON COLUMNS,
    ORDER(
        FILTER(
            NONEMPTY(
                CrossJoin(
                    EXCEPT(
                        [DIM Home Club].[Club ID].[Club ID].MEMBERS,
                        { [DIM Home Club].[Club ID].[Unknown] }
                    ),
                    EXCEPT(
                        [DIM Home Club].[Stadium Name].[Stadium Name].MEMBERS,
                        { [DIM Home Club].[Stadium Name].[Unknown] }
                    )
                ),
                [Measures].[Official Match Count]
            ),
            [Measures].[Stadium Seats Value] > 60000
        ),
        [Measures].[Stadium Seats Value],
        BDESC
    ) ON ROWS
FROM [DW Football Transfermarkt];
